## API Usage Examples

### Trust Accounting
```javascript
import { RealEstateTrustAccounting } from '@/api';

// Create account
const accountId = await RealEstateTrustAccounting.createTrustAccount(
  companyId,
  {
    accountName: 'Main Operating',
    accountType: 'operating',
    bankName: 'First Bank'
  },
  userId
);

// Record transaction
await RealEstateTrustAccounting.recordTrustTransaction(
  companyId,
  {
    trustAccountId: accountId,
    transactionType: 'deposit',
    amount: 50000,
    description: 'Earnest money'
  },
  userId
);
```

### Offers
```javascript
import { RealEstateOffers } from '@/api';

// Create counter-offer
const counterId = await RealEstateOffers.createCounterOffer(
  companyId,
  {
    originalOfferId: 'offer-123',
    counterOfferBy: 'seller',
    counterOfferPrice: 425000
  },
  userId
);
```

### Documents
```javascript
import { RealEstateDocuments } from '@/api';

// Request signature
const sigId = await RealEstateDocuments.requestSignature(
  companyId,
  {
    documentId: 'doc-456',
    signerEmail: 'john@example.com'
  },
  userId
);
```

### Notifications
```javascript
import NotificationService from '@/api/NotificationService';

// Send notification
await NotificationService.notifyOfferCreated(
  companyId,
  { propertyAddress: '123 Main St', offerAmount: 450000 },
  'agent@example.com'
);
```
